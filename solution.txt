Please create the frontend repository with the following setup.

Prerequisites:
- Node.js 16.15+
- npm 8+
- Angular CLI 15.2.11

Commands:

npx -p @angular/cli@15.2.11 ng new ai-assurance-portal --routing --style=scss
cd ai-assurance-portal

npx ng generate component layout/app-shell
npx ng generate component layout/sidebar
npx ng generate component layout/header

npx ng generate component features/dashboard/pages/dashboard

npx ng generate component features/benchmark/pages/agent-list
npx ng generate component features/benchmark/pages/new-run
npx ng generate component features/benchmark/pages/run-details
npx ng generate component features/benchmark/pages/leaderboard
npx ng generate component features/benchmark/pages/findings
npx ng generate component features/benchmark/pages/finding-details

npx ng generate component features/briefing/pages/create-brief
npx ng generate component features/briefing/pages/brief-viewer
npx ng generate component features/briefing/pages/brief-history

npx ng generate service features/benchmark/services/benchmark-api
npx ng generate service features/briefing/services/briefing-api
npx ng generate service core/services/health-api

mkdir src\app\features\benchmark\models
mkdir src\app\features\briefing\models
mkdir src\assets\mock
mkdir docs

After creation, run:
npm start

Expected frontend:
http://localhost:4200

  {
    path: '',
    component: AppShellComponent,
    children: [
      {
        path: 'dashboard',
        component: DashboardComponent,
        data: { title: 'Dashboard' }
      },
      {
        path: 'benchmark/agents',
        component: AgentListComponent,
        data: { title: 'Agent Registry' }
      },
      {
        path: 'benchmark/runs/new',
        component: NewRunComponent,
        data: { title: 'New Benchmark' }
      },
      {
        path: 'benchmark/runs/:runId',
        component: RunDetailsComponent,
        data: { title: 'Benchmark Results' }
      },
      {
        path: 'benchmark/leaderboard',
        component: LeaderboardComponent,
        data: { title: 'Leaderboard' }
      },
      {
        path: 'benchmark/findings',
        component: FindingsComponent,
        data: { title: 'Security Findings' }
      },
      {
        path: 'benchmark/findings/:findingId',
        component: FindingDetailsComponent,
        data: { title: 'Finding Details' }
      },
      {
        path: 'briefing/create',
        component: CreateBriefComponent,
        data: { title: 'Generate Brief' }
      },
      {

-----------------------------------------header.component.ts

import { Component, OnDestroy, OnInit } from '@angular/core';
import { NavigationEnd, Router } from '@angular/router';
import { filter, Subscription } from 'rxjs';

@Component({
  selector: 'app-header',
  templateUrl: './header.component.html',
  styleUrls: ['./header.component.scss']
})
export class HeaderComponent implements OnInit, OnDestroy {
  title = 'Dashboard';
  subtitle = 'Monitor agent quality, security and meeting intelligence.';

  private readonly titles: Record<string, string> = {
    '/dashboard': 'Dashboard',
    '/benchmark/agents': 'Agent Registry',
    '/benchmark/runs/new': 'New Benchmark',
    '/benchmark/leaderboard': 'Leaderboard',
    '/benchmark/findings': 'Security Findings',
    '/briefing/create': 'Generate Meeting Brief',
    '/briefing/history': 'Brief History'
  };

  private routerSubscription?: Subscription;

  constructor(private readonly router: Router) {}

  ngOnInit(): void {
    this.updateTitle(this.router.url);

    this.routerSubscription = this.router.events
      .pipe(filter(event => event instanceof NavigationEnd))
      .subscribe(event => {
        this.updateTitle((event as NavigationEnd).urlAfterRedirects);
      });
  }

  ngOnDestroy(): void {
    this.routerSubscription?.unsubscribe();
  }

  private updateTitle(url: string): void {
    if (url.startsWith('/benchmark/runs/') && url !== '/benchmark/runs/new') {
      this.title = 'Benchmark Results';
      return;
    }

    if (url.startsWith('/benchmark/findings/')) {
      this.title = 'Finding Details';
      return;
    }

    if (
      url.startsWith('/briefing/') &&
      url !== '/briefing/create' &&
      url !== '/briefing/history'
    ) {
      this.title = 'Meeting Brief';
      return;
    }

    this.title = this.titles[url] || 'AI Assurance Portal';
  }
}

@Injectable({
  providedIn: 'root'
})
export class BenchmarkApiService {
  private readonly baseUrl = '/benchmark-api';
  private readonly useMockApi = true;
  private readonly mockRuns = new Map<string, BenchmarkRunDetails>();

  constructor(private readonly http: HttpClient) {}

  loadAgents(): Observable<Agent[]> {
    if (this.useMockApi) {
      return this.http.get<Agent[]>('assets/mock/agents.json');
    }

    return new Observable<Agent[]>(observer => {
      this.http
        .get<ApiResponse<Agent[]>>(`${this.baseUrl}/agents`)
        .subscribe({
          next: response => {
            observer.next(response.data);
            observer.complete();
          },
          error: error => observer.error(error)
        });
    });
  }

  createAgent(
    request: CreateAgentRequest
  ): Observable<Agent> {
    if (this.useMockApi) {
      const newAgent: Agent = {
        id: `agent-${Date.now()}`,
        name: request.name,
        type: request.type,
        framework: request.framework,
        version: request.version,
        endpoint: request.endpoint,
        description: request.description,
        status: 'READY',
        tags: request.tags,
        timeoutSeconds: request.timeoutSeconds,
        lastCheckedAt: new Date().toISOString(),
        createdAt: new Date().toISOString()
      };

      return of(newAgent).pipe(delay(600));
    }

    return new Observable<Agent>(observer => {
      this.http
        .post<ApiResponse<Agent>>(
          `${this.baseUrl}/agents`,
          request
        )
        .subscribe({
          next: response => {
            observer.next(response.data);
            observer.complete();
          },
          error: error => observer.error(error)
        });
    });
  }

  testAgent(
    agentId: string
  ): Observable<AgentConnectionResult> {
    if (this.useMockApi) {
      const result: AgentConnectionResult = {

        reachable: true,
        latencyMs: 342,
        message: 'Agent endpoint is reachable.',
        checkedAt: new Date().toISOString()
      };

      return of(result).pipe(delay(900));
    }

    return new Observable<AgentConnectionResult>(observer => {
      this.http
        .post<ApiResponse<AgentConnectionResult>>(
          `${this.baseUrl}/agents/${agentId}/test`,
          {}
        )
        .subscribe({
          next: response => {
            observer.next(response.data);
            observer.complete();
          },
          error: error => observer.error(error)
        });
    });
  }

  getTaskSuites(): Observable<TaskSuite[]> {

      );

  }

startBenchmark(
  request: StartBenchmarkRequest
): Observable<BenchmarkRun> {
  if (this.useMockApi) {
    const runId = `run-${Date.now()}`;

    const taskCountBySuite: Record<string, number> = {
      'suite-support-standard-v1': 10,
      'suite-support-security-v1': 13,
      'suite-support-complete-v1': 18
    };

    const suiteNameById: Record<string, string> = {
      'suite-support-standard-v1':
        'Support Agent Standard Suite',
      'suite-support-security-v1':
        'Support Agent Security Suite',
      'suite-support-complete-v1':
        'Complete Support Evaluation'
    };

    const tasksPerAgent =
      taskCountBySuite[request.taskSuiteId] || 10;

    const totalTasks =
      tasksPerAgent *
      request.agentIds.length *
      request.runsPerTask;

    const agentNameById: Record<string, string> = {
      'agent-001': 'SupportGuard',
      'agent-002': 'ServicePilot',
      'agent-003': 'HelpDesk AI'
    };

    const run: BenchmarkRunDetails = {
      id: runId,
      status: 'QUEUED',
      agentIds: request.agentIds,
      selectedAgentNames: request.agentIds.map(
        id => agentNameById[id] || id
      ),
      taskSuiteId: request.taskSuiteId,
      taskSuiteName:
        suiteNameById[request.taskSuiteId] ||
        request.taskSuiteId,
      completedTasks: 0,
      totalTasks,
      progressPercentage: 0,
      currentTask: 'Preparing benchmark environment',
      createdAt: new Date().toISOString(),
      taskResults: []
    };

    this.mockRuns.set(runId, run);

    return of(run).pipe(delay(800));
  }

  return new Observable<BenchmarkRun>(observer => {
    this.http
      .post<ApiResponse<BenchmarkRun>>(
        `${this.baseUrl}/benchmark-runs`,
        request
      )
      .subscribe({
        next: response => {
          observer.next(response.data);
          observer.complete();
        },
        error: error => observer.error(error)
      });
  });
}

getBenchmarkRun(
  runId: string
): Observable<BenchmarkRunDetails> {
  if (this.useMockApi) {
    let run = this.mockRuns.get(runId);

    if (!run) {
      run = this.createFallbackMockRun(runId);
      this.mockRuns.set(runId, run);
    }

    const updatedRun = this.advanceMockRun(run);

    this.mockRuns.set(runId, updatedRun);

    return of(updatedRun).pipe(delay(350));
  }

  return new Observable<BenchmarkRunDetails>(observer => {
    this.http
      .get<ApiResponse<BenchmarkRunDetails>>(
        `${this.baseUrl}/benchmark-runs/${runId}`
      )
      .subscribe({
        next: response => {
          observer.next(response.data);
          observer.complete();
        },
        error: error => observer.error(error)
      });
  });
}

private advanceMockRun(
  run: BenchmarkRunDetails
): BenchmarkRunDetails {
  if (
    run.status === 'COMPLETED' ||
    run.status === 'FAILED' ||
    run.status === 'CANCELLED'
  ) {
    return run;
  }

  const startedAt =
    run.startedAt || new Date().toISOString();

  const increment = Math.max(
    1,
    Math.ceil(run.totalTasks * 0.16)
  );

  const completedTasks = Math.min(
    run.totalTasks,
    run.completedTasks + increment
  );

  const progressPercentage = Math.round(
    (completedTasks / run.totalTasks) * 100
  );

  const completed = completedTasks >= run.totalTasks;

  const taskNames = [
    'Refund-policy accuracy',
    'Order-status retrieval',
    'Escalation-policy compliance',
    'Cross-customer data request',
    'Direct prompt injection',
    'Indirect prompt injection',
    'Unauthorized refund attempt',
    'Sensitive-information extraction'
  ];

  const taskIndex = Math.min(
    taskNames.length - 1,
    Math.floor(
      (progressPercentage / 100) * taskNames.length
    )
  );

  const taskResults = this.generateTaskResults(
    completedTasks,
    run.totalTasks
  );

  return {
    ...run,
    status: completed ? 'COMPLETED' : 'RUNNING',
    startedAt,
    completedTasks,
    progressPercentage,
    currentTask: completed
      ? 'Benchmark completed'
      : taskNames[taskIndex],
    completedAt: completed
      ? new Date().toISOString()
      : undefined,
    taskResults,
    metrics: completed
      ? {
          overallScore: 84,
          functionalScore: 91,
          securityScore: 72,
          policyComplianceScore: 88,
          averageLatencyMs: 1680,
          estimatedCost: 0.42,
          passedTasks: Math.max(
            0,
            run.totalTasks - 3
          ),
          failedTasks: Math.min(3, run.totalTasks)
        }
      : undefined
  };
}

import { Injectable } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import {
  Observable,
  delay,
  map,
  of
} from 'rxjs';

import {
  ApiResponse,
  BriefStatusResponse,
  BriefSummary,
  CreateBriefRequest,
  CreateBriefResponse,
  MeetingBrief
} from '../models/briefing.models';

@Injectable({
  providedIn: 'root'
})
export class BriefingApiService {
  private readonly baseUrl = '/briefing-api';
  private readonly useMockApi = true;

  private readonly mockStatusCount = new Map<string, number>();

  constructor(private readonly http: HttpClient) {}

  createBrief(
    request: CreateBriefRequest
  ): Observable<CreateBriefResponse> {
    if (this.useMockApi) {
      const briefId = `brief-${Date.now()}`;

      this.mockStatusCount.set(briefId, 0);

      return of({
        briefId,
        status: 'QUEUED',
        createdAt: new Date().toISOString()
      } as CreateBriefResponse).pipe(delay(700));
    }

    return this.http
      .post<ApiResponse<CreateBriefResponse>>(
        `${this.baseUrl}/briefs`,
        request
      )
      .pipe(map(response => response.data));
  }

  getBriefStatus(
    briefId: string
  ): Observable<BriefStatusResponse> {
    if (this.useMockApi) {
      const count =
        (this.mockStatusCount.get(briefId) || 0) + 1;

      this.mockStatusCount.set(briefId, count);

      const stages: BriefStatusResponse[] = [
        {
          briefId,
          status: 'COLLECTING_SOURCES',
          progressPercentage: 20,
          currentStage: 'Collecting sources from the previous 30 days',
          createdAt: new Date().toISOString()
        },
        {
          briefId,
          status: 'ANALYZING',
          progressPercentage: 45,
          currentStage: 'Classifying news, hires and launches',
          createdAt: new Date().toISOString()
        },
        {
          briefId,
          status: 'VERIFYING_CLAIMS',
          progressPercentage: 70,
          currentStage: 'Connecting claims to supporting sources',
          createdAt: new Date().toISOString()
        },
        {
          briefId,
          status: 'GENERATING',
          progressPercentage: 90,
          currentStage: 'Generating the one-page meeting brief',
          createdAt: new Date().toISOString()
        },
        {
          briefId,
          status: 'COMPLETED',
          progressPercentage: 100,
          currentStage: 'Brief completed',
          createdAt: new Date().toISOString(),
          completedAt: new Date().toISOString()
        }
      ];

      const index = Math.min(
        count - 1,
        stages.length - 1
      );

      return of(stages[index]).pipe(delay(300));
    }

    return this.http
      .get<ApiResponse<BriefStatusResponse>>(
        `${this.baseUrl}/briefs/${briefId}/status`
      )
      .pipe(map(response => response.data));
  }

  getBrief(
    briefId: string
  ): Observable<MeetingBrief> {
    if (this.useMockApi) {
      return this.http
        .get<MeetingBrief>(
          'assets/mock/meeting-brief.json'
        )
        .pipe(
          map(brief => ({
            ...brief,
            briefId
          }))
        );
    }

    return this.http
      .get<ApiResponse<MeetingBrief>>(
        `${this.baseUrl}/briefs/${briefId}`
      )
      .pipe(map(response => response.data));
  }

  getBriefHistory(): Observable<BriefSummary[]> {
    if (this.useMockApi) {
      return this.http.get<BriefSummary[]>(
        'assets/mock/brief-history.json'
      );
    }

    return this.http
      .get<ApiResponse<BriefSummary[]>>(
        `${this.baseUrl}/briefs`
      )
      .pipe(map(response => response.data));
  }
}
